1. هيكل الملفات المهم

الملفات الأساسية اللي راح تعدل عليها:

resources/ ├── css/ │ └── app.css ← كل الألوان والمتغيرات العامة ├── js/ │ ├── pages/ ← صفحات الموقع (Dashboard, Transactions, ...) │ │ ├── Dashboard.svelte │ │ ├── Transactions.svelte │ │ ├── AiAssistant.svelte │ │ ├── BudgetGoals.svelte │ │ └── Profile.svelte │ ├── layouts/ ← تخطيطات الصفحات │ │ └── app/MainAppLayout.svelte │ └── components/ ← مكونات مشتركة │ ├── DesktopTopNav.svelte ← شريط التنقل العلوي (لابتوب) │ ├── MobileBottomNav.svelte ← شريط التنقل السفلي (جوال) │ ├── AppShell.svelte │ └── ui/ ← مكونات UI أساسية (أزرار، بطاقات، ...)
ملاحظة: المشروع يستخدم TailwindCSS v4 — الألوان تُعرّف كـ CSS variables في app.css وتُستخدم كـ utility classes في ملفات Svelte.

2. تعديل الألوان

كل ألوان الموقع مُعرّفة كـ CSS custom properties (متغيرات) في ملف:

resources/css/app.css

أين تُعرّف الألوان؟

في قسم :root (للوضع الفاتح) وقسم .dark (للوضع الداكن):

/* الوضع الفاتح - Light Mode */
:root {
    --background: hsl(0 0% 100%);      /* خلفية الصفحة */
    --foreground: hsl(0 0% 3.9%);      /* لون النص */
    --primary: hsl(0 0% 9%);           /* اللون الأساسي (الأزرار، الروابط) */
    --primary-foreground: hsl(0 0% 98%); /* نص فوق اللون الأساسي */
    --secondary: hsl(0 0% 92.1%);      /* اللون الثانوي */
    --muted: hsl(0 0% 96.1%);          /* خلفية مكتومة */
    --muted-foreground: hsl(0 0% 45.1%); /* نص مكتوم */
    --accent: hsl(0 0% 96.1%);         /* لون التمييز */
    --destructive: hsl(0 84.2% 60.2%); /* لون التحذير/الحذف (أحمر) */
    --border: hsl(0 0% 92.8%);         /* لون الحدود */
    --ring: hsl(0 0% 3.9%);            /* لون حلقة التركيز */
    /* ... المزيد */
}

/* الوضع الداكن - Dark Mode */
.dark {
    --background: hsl(0 0% 3.9%);
    --foreground: hsl(0 0% 98%);
    --primary: hsl(0 0% 98%);
    /* ... */
}

كيف أغيّر لون الموقع؟

مثال 1: تغيير اللون الأساسي إلى أزرق:

/* في :root (الفاتح) */
--primary: hsl(220 90% 56%);           /* أزرق */
--primary-foreground: hsl(0 0% 100%);  /* أبيض للنص فوقه */

/* في .dark (الداكن) */
--primary: hsl(220 90% 60%);           /* أزرق أفتح شوي للداكن */
--primary-foreground: hsl(0 0% 100%);

مثال 2: تغيير لون الخلفية إلى رمادي فاتح:

:root {
    --background: hsl(210 20% 98%);    /* رمادي فاتح جداً مع مسحة زرقاء */
}
.dark {
    --background: hsl(220 15% 8%);     /* رمادي داكن للوضع الليلي */
}

فهم صيغة HSL

hsl(220 90% 56%)
220 = درجة اللون (أزرق)
90% = التشبع
56% = السطوع
hsl(0 84% 60%)
0 = درجة اللون (أحمر)
84% = التشبع
60% = السطوع
hsl(150 60% 45%)
150 = درجة اللون (أخضر)
60% = التشبع
45% = السطوع
درجات اللون (Hue): 0=أحمر، 30=برتقالي، 60=أصفر، 120=أخضر، 180=سماوي، 220=أزرق، 270=بنفسجي، 330=وردي

ألوان الرسوم البيانية (Charts)

--chart-1: hsl(12 76% 61%);    /* برتقالي */
--chart-2: hsl(173 58% 39%);   /* أخضر مائل للأزرق */
--chart-3: hsl(197 37% 24%);   /* أزرق داكن */
--chart-4: hsl(43 74% 66%);    /* ذهبي */
--chart-5: hsl(27 87% 67%);    /* برتقالي فاتح */

3. تعديل الأحجام والأبعاد

الأحجام تُتحكم فيها عبر Tailwind utility classes مباشرة في ملفات Svelte.

أحجام النصوص

<!-- نص صغير -->
<span class="text-xs">       /* 0.75rem = 12px */
<span class="text-sm">       /* 0.875rem = 14px */
<span class="text-base">     /* 1rem = 16px (الافتراضي) */
<span class="text-lg">       /* 1.125rem = 18px */
<span class="text-xl">       /* 1.25rem = 20px */
<span class="text-2xl">      /* 1.5rem = 24px */
<span class="text-3xl">      /* 1.875rem = 30px */

أحجام العناصر (padding, margin, gap)

<!-- المسافات الداخلية (padding) -->
<div class="p-4">       /* padding: 1rem = 16px من كل الجهات */
<div class="px-6">      /* padding-left/right: 1.5rem = 24px */
<div class="py-2">      /* padding-top/bottom: 0.5rem = 8px */

<!-- المسافات الخارجية (margin) -->
<div class="m-4">       /* margin: 1rem من كل الجهات */
<div class="mb-6">      /* margin-bottom: 1.5rem */
<div class="mt-2">      /* margin-top: 0.5rem */

<!-- المسافات بين العناصر (gap) -->
<div class="flex gap-2">  /* مسافة 0.5rem بين العناصر */
<div class="flex gap-4">  /* مسافة 1rem بين العناصر */

تعديل حجم عنصر محدد

مثال: تكبير زر في MobileBottomNav.svelte:

<!-- قبل -->
<div class="flex size-14 -mt-5 items-center justify-center rounded-full">

<!-- بعد (تكبير الزر) -->
<div class="flex size-16 -mt-6 items-center justify-center rounded-full">

تعديل عرض العناصر

<!-- عرض ثابت -->
<div class="w-64">          /* عرض 16rem = 256px */
<div class="w-full">        /* عرض 100% */
<div class="w-screen">      /* عرض الشاشة كاملة */

<!-- عرض نسبي (responsive) -->
<div class="w-full md:w-1/2 lg:w-1/3">  /* جوال=كامل، تابلت=نص، لابتوب=ثلث */

تعديل ارتفاع العناصر

<div class="h-16">          /* ارتفاع 4rem = 64px */
<div class="h-screen">      /* ارتفاع الشاشة (لا تستخدمه - استخدم min-h-[100dvh]) */
<div class="min-h-[100dvh]"> /* الحد الأدنى = ارتفاع الشاشة (هذا هو الصحيح) */

الزوايا الدائرية (Border Radius)

<div class="rounded-sm">    /* 4px */
<div class="rounded-md">    /* 6px */
<div class="rounded-lg">    /* 8px (الافتراضي) */
<div class="rounded-xl">    /* 12px */
<div class="rounded-2xl">   /* 16px */
<div class="rounded-full">  /* دائرة كاملة */

4. إضافة تأثيرات CSS

تأثيرات الانتقال (Transitions)

<!-- انتقال اللون عند التمرير -->
<button class="bg-blue-500 transition-colors hover:bg-blue-600">

<!-- انتقال مع تحريك -->
<div class="transition-all duration-300 hover:scale-105 hover:shadow-lg">

<!-- انتقال سريع -->
<div class="transition-transform duration-200 hover:-translate-y-1">

تأثير الظل (Shadow)

<div class="shadow-sm">     /* ظل خفيف */
<div class="shadow-md">     /* ظل متوسط */
<div class="shadow-lg">     /* ظل كبير */
<div class="shadow-xl">     /* ظل كبير جداً */

/* ظل مخصص */
<div class="shadow-[0_4px_20px_rgba(59,130,246,0.3)]">  /* ظل أزرق */

تأثير الزجاج (Glassmorphism) - موجود بالمشروع

<!-- هذا التأثير موجود في شريط التنقل -->
<nav class="bg-background/95 backdrop-blur-lg border-b border-border">

/* الشفافية: bg-background/95 = خلفية بشفافية 95% */
/* الضبابية: backdrop-blur-lg = تأثير زجاج ضبابي */

تأثيرات hover مخصصة

<!-- تكبير خفيف مع ظل -->
<div class="transition-all duration-200 hover:scale-[1.03] hover:shadow-[0_8px_24px_rgba(0,0,0,0.12)]">

<!-- تأثير الضغط -->
<button class="active:scale-95 active:translate-y-px">

<!-- توهج (glow) -->
<div class="hover:shadow-[0_0_20px_rgba(59,130,246,0.4)]">

تأثيرات الحركة (Animations)

<!-- ظهور تدريجي -->
<div class="animate-in fade-in duration-300">

<!-- انزلاق من الأسفل -->
<div class="animate-in slide-in-from-bottom-4 duration-500">

<!-- دوران -->
<div class="animate-spin">

/* حركة مخصصة - أضفها في app.css */
@keyframes pulse-glow {
    0%, 100% { box-shadow: 0 0 0 0 rgba(59,130,246,0.4); }
    50% { box-shadow: 0 0 20px 4px rgba(59,130,246,0.2); }
}
.pulse-glow { animation: pulse-glow 2s ease-in-out infinite; }

تأثيرات Gradient

<!-- تدرج لوني -->
<div class="bg-gradient-to-r from-blue-500 to-purple-600">

<!-- تدرج مع نص -->
<h1 class="bg-gradient-to-r from-blue-400 to-emerald-400 bg-clip-text text-transparent">

/* تدرج مخصص في app.css */
.glass-gradient {
    background: linear-gradient(135deg,
        rgba(255,255,255,0.1) 0%,
        rgba(255,255,255,0.05) 100%
    );
}

5. الاستجابة للشاشات (Responsive Design)

المشروع يستخدم نظام Mobile-First — الكلاس يُطبق على الجوال أولاً، ثم تُضيف breakpoint للشاشات الأكبر.

Nقاط التوقف (Breakpoints)

البادئة العرض الأدنى الجهاز مثال
(بدون) 0px جوال class="hidden" = مخفي دائماً
sm: 640px جوال كبير class="sm:hidden" = مخفي من 640px+
md: 768px تابلت class="md:block" = ظاهر من 768px+
lg: 1024px لابتوب class="lg:hidden" = مخفي من 1024px+
xl: 1280px شاشة كبيرة class="xl:flex" = flex من 1280px+

كيف يعمل النظام؟

<!-- مثال: عنصر مخفي في الجوال، ظاهر في اللابتوب -->
<div class="hidden lg:block">محتوى اللابتوب فقط</div>

<!-- مثال: عنصر ظاهر في الجوال، مخفي في اللابتوب -->
<div class="block lg:hidden">محتوى الجوال فقط</div>

<!-- مثال: حجم نص يتغير حسب الشاشة -->
<h1 class="text-xl md:text-2xl lg:text-3xl">عنوان متجاوب</h1>

<!-- مثال: أعمدة تتغير حسب الشاشة -->
<div class="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
    <!-- جوال: عمود واحد | تابلت: عمودين | لابتوب: 3 أعمدة -->
</div>
القاعدة الأساسية: اكتب الكلاس للجوال أولاً (بدون بادئة)، ثم أضف lg: للابتوب. مثال: class="p-4 lg:p-6" يعني padding صغير في الجوال، أكبر في اللابتوب.
